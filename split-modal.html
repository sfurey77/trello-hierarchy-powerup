Initialize iframe with API credentials const t = TrelloPowerUp.iframe({ appKey: API_KEY, appName: 'Task Hierarchy Power-Up' }); t.render(function () { const listContainer = document.getElementById('item-list'); return t.card('id', 'name', 'idList') .then(function (parentCard) { // Check API Authorization Status return t.getRestApi() .isAuthorized() .then(function (isAuthorized) { if (!isAuthorized) { listContainer.innerHTML = ''; t.sizeTo('#content'); document.getElementById('auth-btn').onclick = function() { t.getRestApi().authorize().then(function() { location.reload(); }).catch(function(err) { console.error("Auth error:", err); }); }; return; } // User is Authorized -> Fetch Token & Get Checklists return t.getRestApi().getToken().then(function (token) { if (!token) { listContainer.innerHTML = '

Authorization token missing. Click authorize again.

'; return t.sizeTo('#content'); } return fetch(`https://api.trello.com/1/cards/${parentCard.id}/checklists?key=${API_KEY}&token=${token}`) .then(res => res.json()) .then(function (checklists) { listContainer.innerHTML = ''; // Clear loading text if (!checklists || checklists.length === 0) { listContainer.innerHTML = '

No checklists found on this card. Add a checklist to get started!

'; return t.sizeTo('#content'); } let hasIncomplete = false; checklists.forEach(function (checklist) { const checkItems = checklist.checkItems || []; checkItems.forEach(function (item) { if (item.state === 'incomplete') { hasIncomplete = true; const btn = document.createElement('button'); btn.className = 'item-btn'; btn.textContent = `Split: ${item.name}`; btn.onclick = function () { btn.disabled = true; btn.textContent = 'Creating Child Card...'; createChildCard(item, parentCard, token); }; listContainer.appendChild(btn); } }); }); if (!hasIncomplete) { listContainer.innerHTML = '

All checklist tasks are completed!

'; } return t.sizeTo('#content'); }); }); }); }) .catch(function (err) { console.error("SDK Error:", err); listContainer.innerHTML = '

Error reading checklists. Please check browser console or re-authorize.

'; t.sizeTo('#content'); }); }); function createChildCard(itemData, parentCard, token) { fetch(`https://api.trello.com/1/cards?key=${API_KEY}&token=${token}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: itemData.name, idList: parentCard.idList, desc: `**Parent Project:** [${parentCard.name}](https://trello.com/c/${parentCard.id})` }) }) .then(res => res.json()) .then(function (childCard) { return t.set(childCard.id, 'shared', 'parentDetails', { parentId: parentCard.id, checkitemId: itemData.id }) .then(function () { return fetch(`https://api.trello.com/1/cards/${parentCard.id}/checkItem/${itemData.id}?key=${API_KEY}&token=${token}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: `${itemData.name} 🔗 [Child Card](${childCard.shortUrl})` }) }); }); }) .then(function () { location.reload(); }) .catch(function (err) { console.error("Error creating child card:", err); alert("Failed to create child card."); }); }
